# Per-Region Calibration and Map Wiring Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fit a per-region coordinate transform from the 117 calibration anchors and use it to
replace those locations' hand-guessed `fx`/`fy` with calibrated values in both copies of
`locations.ts`.

**Architecture:** A pure affine-fit module (least squares, closed-form) computes one transform
per surface region; a disposable script applies it and rewrites the affected lines in place.

**Tech Stack:** TypeScript, `node:test` (this project's established test convention).

## Global Constraints

- Only the 11 known surface regions get calibrated (`consecrated-snowfield`, `haligtree`,
  `mountaintops`, `farum-azula`, `mt-gelmir`, `altus-plateau`, `leyndell`, `liurnia`, `caelid`,
  `limgrave`, `weeping-peninsula`) — underground regions (`nokron`, `ainsel-river`, `nokstella`,
  `deeproot-depths`, `lake-of-rot`, `siofra-river`, `mohgwyn-palace`) are explicitly out of
  scope and must be filtered out before fitting, since several have fewer than 3 anchors
  (`ainsel-river`: 2, `deeproot-depths`: 2, `nokstella`: 1, `lake-of-rot`: 1) and would otherwise
  crash the "fewer than 3 points" check.
- Exact-name matching only when rewriting `locations.ts` — never fuzzy/normalized — since a
  fuzzy match risks silently rewriting the wrong line.
- Both `web/src/map/locations.ts` and `client/src/renderer/map/locations.ts` get the same edits
  (confirmed byte-for-byte identical before this work starts).

The fixture points below were generated by hand-applying a known transform
(`a=2, b=0.5, c=1, d=-0.3, e=1.5, f=2`) to four arbitrary world coordinates — e.g. for

```typescript
// server/src/pipeline/affine-fit.test.ts
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { fitAffineTransform, applyAffineTransform, type AffinePoint } from './affine-fit.js';

// Generated by hand from the known transform a=2, b=0.5, c=1, d=-0.3, e=1.5, f=2.
const THREE_POINTS: AffinePoint[] = [
  { worldX: 0, worldZ: 0, fx: 1, fy: 2 },
  { worldX: 100, worldZ: 0, fx: 201, fy: -28 },
  { worldX: 0, worldZ: 100, fx: 51, fy: 152 },
];

const FOUR_POINTS: AffinePoint[] = [
  ...THREE_POINTS,
  { worldX: 50, worldZ: 50, fx: 126, fy: 62 },
];

function assertRecoversKnownTransform(transform: ReturnType<typeof fitAffineTransform>) {
  assert.ok(Math.abs(transform.a - 2) < 1e-9);
  assert.ok(Math.abs(transform.b - 0.5) < 1e-9);
  assert.ok(Math.abs(transform.c - 1) < 1e-9);
  assert.ok(Math.abs(transform.d - -0.3) < 1e-9);
  assert.ok(Math.abs(transform.e - 1.5) < 1e-9);
  assert.ok(Math.abs(transform.f - 2) < 1e-9);
}

test('fitAffineTransform recovers an exact transform from exactly 3 points', () => {
  assertRecoversKnownTransform(fitAffineTransform(THREE_POINTS));
});

test('fitAffineTransform recovers an exact transform from more than 3 consistent points', () => {
  assertRecoversKnownTransform(fitAffineTransform(FOUR_POINTS));
});

test('fitAffineTransform throws with fewer than 3 points', () => {
  assert.throws(() => fitAffineTransform(THREE_POINTS.slice(0, 2)), /at least 3 points/);
});

test('applyAffineTransform round-trips through a fitted transform', () => {
  const transform = fitAffineTransform(THREE_POINTS);
  const result = applyAffineTransform(transform, 50, 50);
  assert.ok(Math.abs(result.fx - 126) < 1e-9);
  assert.ok(Math.abs(result.fy - 62) < 1e-9);
});
```

```typescript
// server/src/pipeline/affine-fit.ts
export interface AffinePoint {
  worldX: number;
  worldZ: number;
  fx: number;
  fy: number;
}

export interface AffineTransform {
  a: number;
  b: number;
  c: number;
  d: number;
  e: number;
  f: number;
}

function det3x3(m: number[][]): number {
  return (
    m[0][0] * (m[1][1] * m[2][2] - m[1][2] * m[2][1]) -
    m[0][1] * (m[1][0] * m[2][2] - m[1][2] * m[2][0]) +
    m[0][2] * (m[1][0] * m[2][1] - m[1][1] * m[2][0])
  );
}

// Solves the 3x3 linear system M*v = b via Cramer's rule.
function solve3x3(m: number[][], b: number[]): number[] {
  const d = det3x3(m);
  if (Math.abs(d) < 1e-9) {
    throw new Error('Cannot fit an affine transform: points are collinear or coincident');
  }

  const withColumn = (col: number) =>
    m.map((row, i) => row.map((val, j) => (j === col ? b[i] : val)));

  return [0, 1, 2].map((col) => det3x3(withColumn(col)) / d);
}

/**
 * Fits fx = a*worldX + b*worldZ + c and fy = d*worldX + e*worldZ + f via least squares
 * (normal equations, solved in closed form). With exactly 3 points the system is exactly
 * determined (zero residual); with more, it's the standard least-squares solution.
 */
export function fitAffineTransform(points: AffinePoint[]): AffineTransform {
  if (points.length < 3) {
    throw new Error(`fitAffineTransform needs at least 3 points, got ${points.length}`);
  }

  let sxx = 0, sxz = 0, sx = 0, szz = 0, sz = 0;
  let sxfx = 0, szfx = 0, sfx = 0;
  let sxfy = 0, szfy = 0, sfy = 0;

  for (const p of points) {
    sxx += p.worldX * p.worldX;
    sxz += p.worldX * p.worldZ;
    sx += p.worldX;
    szz += p.worldZ * p.worldZ;
    sz += p.worldZ;
    sxfx += p.worldX * p.fx;
    szfx += p.worldZ * p.fx;
    sfx += p.fx;
    sxfy += p.worldX * p.fy;
    szfy += p.worldZ * p.fy;
    sfy += p.fy;
  }

  const n = points.length;
  const m = [
    [sxx, sxz, sx],
    [sxz, szz, sz],
    [sx, sz, n],
  ];

  const [a, b, c] = solve3x3(m, [sxfx, szfx, sfx]);
  const [d, e, f] = solve3x3(m, [sxfy, szfy, sfy]);

  return { a, b, c, d, e, f };
}

export function applyAffineTransform(
  transform: AffineTransform,
  worldX: number,
  worldZ: number,
): { fx: number; fy: number } {
  const { a, b, c, d, e, f } = transform;
  return {
    fx: a * worldX + b * worldZ + c,
    fy: d * worldX + e * worldZ + f,
  };
}
```

```typescript
// server/scripts/apply-calibration.ts (disposable - delete after running)
import { readFileSync, writeFileSync } from 'node:fs';
import { fitAffineTransform, applyAffineTransform, type AffinePoint } from '../src/pipeline/affine-fit.js';

interface Anchor {
  name: string;
  category: string;
  regionId: string;
  existingFx: number;
  existingFy: number;
  worldX: number;
  worldZ: number;
}

// Only these 11 regions are calibrated - underground regions (nokron, ainsel-river, nokstella,
// deeproot-depths, lake-of-rot, siofra-river, mohgwyn-palace) are out of scope per the design
// spec, and several have fewer than 3 anchors, which would otherwise crash the fit.
const SURFACE_REGION_IDS = new Set([
  'consecrated-snowfield',
  'haligtree',
  'mountaintops',
  'farum-azula',
  'mt-gelmir',
  'altus-plateau',
  'leyndell',
  'liurnia',
  'caelid',
  'limgrave',
  'weeping-peninsula',
]);

const allAnchors: Anchor[] = JSON.parse(
  readFileSync('tools/msb-extractor/output/calibration-anchors.json', 'utf-8'),
);
const anchors = allAnchors.filter((a) => SURFACE_REGION_IDS.has(a.regionId));
console.log(`Calibrating ${anchors.length} of ${allAnchors.length} anchors (surface regions only)`);

const anchorsByRegion = new Map<string, Anchor[]>();
for (const a of anchors) {
  const list = anchorsByRegion.get(a.regionId) ?? [];
  list.push(a);
  anchorsByRegion.set(a.regionId, list);
}

interface Update {
  name: string;
  fx: number;
  fy: number;
}

const updates: Update[] = [];
for (const [regionId, regionAnchors] of anchorsByRegion) {
  const points: AffinePoint[] = regionAnchors.map((a) => ({
    worldX: a.worldX,
    worldZ: a.worldZ,
    fx: a.existingFx,
    fy: a.existingFy,
  }));
  const transform = fitAffineTransform(points);
  for (const a of regionAnchors) {
    const { fx, fy } = applyAffineTransform(transform, a.worldX, a.worldZ);
    updates.push({ name: a.name, fx, fy });
  }
  console.log(`  ${regionId}: fit from ${regionAnchors.length} anchors`);
}

// Extracts a `name: '...'` or `name: "..."` field's unescaped value from one line of
// locations.ts, or null if the line has no name field. Handles both quoting styles present in
// the file - some entries escape an apostrophe inside single quotes ('Oridys\'s Rise'), others
// switch to double quotes instead ("Ranni's Rise") - confirmed by reading the real file.
function extractNameFromLine(line: string): string | null {
  const singleQuoted = /name:\s*'((?:\\'|[^'])*)'/.exec(line);
  if (singleQuoted) return singleQuoted[1].replace(/\\'/g, "'");
  const doubleQuoted = /name:\s*"((?:\\"|[^"])*)"/.exec(line);
  if (doubleQuoted) return doubleQuoted[1].replace(/\\"/g, '"');
  return null;
}

function applyUpdatesToFile(path: string): void {
  const lines = readFileSync(path, 'utf-8').split('\n');
  const updatesByName = new Map(updates.map((u) => [u.name, u]));
  const foundNames = new Set<string>();

  const newLines = lines.map((line) => {
    const name = extractNameFromLine(line);
    if (name === null) return line;
    const update = updatesByName.get(name);
    if (!update) return line;

    foundNames.add(name);
    return line
      .replace(/fx:\s*-?[\d.]+/, `fx: ${update.fx}`)
      .replace(/fy:\s*-?[\d.]+/, `fy: ${update.fy}`);
  });

  const missing = updates.filter((u) => !foundNames.has(u.name));
  if (missing.length > 0) {
    throw new Error(
      `${missing.length} anchor name(s) not found as a line in ${path}: ${missing.map((u) => u.name).join(', ')}`,
    );
  }

  writeFileSync(path, newLines.join('\n'));
  console.log(`Updated ${foundNames.size} entries in ${path}`);
}

applyUpdatesToFile('web/src/map/locations.ts');
applyUpdatesToFile('client/src/renderer/map/locations.ts');
```
